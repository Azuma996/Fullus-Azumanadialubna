    // Persistensi Data LocalStorage

    function loadData() {

      const storedTx = localStorage.getItem('full_us_transactions');

      if (storedTx) {

        try {

          transactions = JSON.parse(storedTx);

        } catch (e) {

          transactions = INITIAL_TRANSACTIONS;

        }

      } else {

        transactions = INITIAL_TRANSACTIONS;

        saveTransactionsToStorage();

      }

      const storedBudgets = localStorage.getItem('full_us_budgets');

      if (storedBudgets) {

        try {

          budgets = JSON.parse(storedBudgets);

        } catch (e) {

          budgets = DEFAULT_BUDGETS;

        }

      } else {

        budgets = DEFAULT_BUDGETS;

        saveBudgetsToStorage();

      }

    }

    function saveTransactionsToStorage() {

    function formatRupiah(amount) {

      return new Intl.NumberFormat('id-ID', {

        style: 'currency',

        currency: 'IDR',

        maximumFractionDigits: 0

      }).format(amount);

    }

    function getSelectedMonth() {

      return document.getElementById('globalMonthFilter').value;

    }

    function getFilteredMonthTransactions() {

      const selectedMonth = getSelectedMonth();

      return transactions.filter(t => t.date && t.date.startsWith(selectedMonth));

    }

    function renderSummaryCards() {

      const currentMonthTx = getFilteredMonthTransactions();

     

      let incomeTotal = 0;

      let incomeCount = 0;

      let expenseTotal = 0;

      let expenseCount = 0;

      currentMonthTx.forEach(t => {

        const val = parseFloat(t.amount) || 0;

        if (t.type === 'INCOME') {

          incomeTotal += val;

          incomeCount++;

        } else {

          expenseTotal += val;

          expenseCount++;

        }

      });

      const balance = incomeTotal - expenseTotal;

      document.getElementById('totalBalance').innerText = formatRupiah(balance);

      document.getElementById('totalIncome').innerText = formatRupiah(incomeTotal);

      document.getElementById('totalExpense').innerText = formatRupiah(expenseTotal);

      document.getElementById('incomeCountNote').innerText = `${incomeCount} transaksi masuk`;

      document.getElementById('expenseCountNote').innerText = `${expenseCount} transaksi keluar`;

      const balanceEl = document.getElementById('totalBalance');

      if (balance < 0) {

        balanceEl.className = 'text-2xl sm:text-3xl font-extrabold text-rose-600';

        document.getElementById('balanceStatusNote').innerText = 'Peringatan: Pengeluaran melampaui pemasukan!';

      } else {

        balanceEl.className = 'text-2xl sm:text-3xl font-extrabold text-slate-900';

        document.getElementById('balanceStatusNote').innerText = 'Keluarga dalam kondisi surplus (Aman)';

      }

    }

    function renderBudgetProgress() {

      const container = document.getElementById('budgetContainer');

      container.innerHTML = '';

      const currentMonthTx = getFilteredMonthTransactions();

      const expensesByCategory = {};

     

      currentMonthTx.forEach(t => {

        if (t.type === 'EXPENSE') {

          expensesByCategory[t.category] = (expensesByCategory[t.category] || 0) + parseFloat(t.amount);

        }

      });

      const categoriesWithBudget = Object.keys(budgets).filter(c => budgets[c] > 0);

      if (categoriesWithBudget.length === 0) {

        container.innerHTML = `

          <div class="text-center py-6 text-slate-400 text-xs">

            Belum ada kuota anggaran. Klik tombol <b>"Atur Kuota"</b> di atas.

          </div>

        `;

        return;

      }

      categoriesWithBudget.forEach(catId => {

        const budgetAmount = budgets[catId];

        const spent = expensesByCategory[catId] || 0;

        const percentage = Math.min(Math.round((spent / budgetAmount) * 100), 100);

        const isOver = spent > budgetAmount;

        const catMeta = EXPENSE_CATEGORIES.find(c => c.id === catId) || { icon: 'fa-tag', name: catId };

        let barColor = 'bg-emerald-500';

        let badge = `<span class="text-[11px] font-semibold text-slate-500">${percentage}%</span>`;

        if (percentage >= 80 && !isOver) {

          barColor = 'bg-amber-500';

        } else if (isOver) {

          barColor = 'bg-rose-500';

          badge = `<span class="text-[11px] font-bold text-rose-600 bg-rose-50 px-1.5 py-0.5 rounded">Over Kuota!</span>`;

        }

        const div = document.createElement('div');

        div.className = 'text-xs';

        div.innerHTML = `

          <div class="flex items-center justify-between mb-1">

            <div class="flex items-center gap-1.5 font-semibold text-slate-700 truncate max-w-[200px]">

              <i class="fa-solid ${catMeta.icon} text-slate-400 text-[11px]"></i>

              <span>${catMeta.name}</span>

            </div>

            <div class="flex items-center gap-2">

              <span class="text-slate-500">${formatRupiah(spent)} <span class="text-slate-300">/</span> ${formatRupiah(budgetAmount)}</span>

              ${badge}

            </div>

          </div>

          <div class="w-full bg-slate-100 rounded-full h-2 overflow-hidden">

            <div class="${barColor} h-2 rounded-full transition-all duration-300" style="width: ${percentage}%"></div>

          </div>

        `;

        container.appendChild(div);

      });

    }

    function renderTransactions() {

      const tbody = document.getElementById('transactionList');

      const emptyEl = document.getElementById('emptyTransactionNotice');

      tbody.innerHTML = '';

      const selectedMonth = getSelectedMonth();

      const filterCategory = document.getElementById('filterCategory').value;

      const filterType = document.getElementById('filterType').value;

      let filtered = transactions.filter(t => {

        const matchesMonth = t.date && t.date.startsWith(selectedMonth);

        const matchesMember = filterMember === 'ALL' || t.member === filterMember;

        const matchesCategory = filterCategory === 'ALL' || t.category === filterCategory;

        const matchesType = filterType === 'ALL' || t.type === filterType;

        return matchesMonth && matchesMember && matchesCategory && matchesType;

    function formatDateDisplay(dateStr) {

      if (!dateStr) return '-';

      const parts = dateStr.split('-');

      if (parts.length < 3) return dateStr;

      const d = new Date(parts[0], parts[1] - 1, parts[2]);

      return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });

    }

    function escapeHtml(str) {

      if (!str) return '';

      return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

    }

    function handleSaveTransaction(e) {

      e.preventDefault();

      const id = document.getElementById('txId').value;

      const type = document.getElementById('txType').value;

      const amount = parseFloat(document.getElementById('txAmount').value);

      const date = document.getElementById('txDate').value;

      const category = document.getElementById('txCategory').value;

      const member = document.getElementById('txMember').value;

      const description = document.getElementById('txDescription').value.trim();

      if (!amount || amount <= 0) return;

      if (id) {

        const index = transactions.findIndex(t => t.id === id);

        if (index !== -1) {

          transactions[index] = { id, type, amount, date, category, member, description };

        }

      } else {

        const newTx = {

          id: 'tx_fullus_' + Date.now(),

          type,

          amount,

          date,

          category,

          member,

          description

        };

        transactions.unshift(newTx);

      }

      saveTransactionsToStorage();

      closeTransactionModal();

      refreshDashboard();

    }

      if (window.confirm("Hapus catatan transaksi ini dari FULL-US?")) {

        transactions = transactions.filter(t => t.id !== id);

        refreshDashboard();

      }

    }

    function exportToCSV() {

      const selectedMonth = getSelectedMonth();

      const currentMonthTx = getFilteredMonthTransactions();

      if (currentMonthTx.length === 0) {

        alert("Tidak ada transaksi untuk diekspor pada bulan ini.");

        return;

      }

      let csvContent = "data:text/csv;charset=utf-8,";

      csvContent += "ID,Tanggal,Jenis,Kategori,Anggota Keluarga,Keterangan,Jumlah (Rp)\r\n";

      currentMonthTx.forEach(t => {

        const typeStr = t.type === 'INCOME' ? 'Pemasukan' : 'Pengeluaran';

        const descEscaped = `"${(t.description || '').replace(/"/g, '""')}"`;

        const row = [

          t.id,

          t.date,

          typeStr,

          `"${t.category}"`,

          t.member,

          descEscaped,

          t.amount

        ].join(",");

        csvContent += row + "\r\n";

      });

      const encodedUri = encodeURI(csvContent);

      const link = document.createElement("a");

      link.setAttribute("href", encodedUri);

      link.setAttribute("download", `FULL-US_Laporan_Keuangan_${selectedMonth}.csv`);

      document.body.appendChild(link);

      link.click();

      document.body.removeChild(link);

    }
